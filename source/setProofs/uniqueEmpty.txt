## Proof that 
### 
### the empty set is unique.

1. Forall X, (forall y, Not(In(y, X))) -> Equals(X, EmptySet). (Universal Generalization from subproof below)
begin subproof
 1.1. Let X be any set. (Arbitrary)
 1.2. Assume: forall y, Not(In(y, X)). (Assumption for Conditional Introduction)
 1.3. Forall y, (In(y, EmptySet) -> In(y, X)). (Universal Generalization from subproof below)
 begin subproof
  1.3.1. Let y be any object. (Arbitrary)
  1.3.2. In(y, EmptySet) -> In(y, X). (Conditional Introduction from subproof below)
  begin subproof
   1.3.2.1. In(y, EmptySet). (Assumption for Conditional Introduction)
   1.3.2.2. Not(In(y, EmptySet)). (EmptySetProperty from 1.3.2.1)
   1.3.2.3. In(y, X). (Explosion from 1.3.2.1, 1.3.2.2)
  end subproof
 end subproof
 1.4. EmptySet is a subset of X. (Universal Instantiation from 1.3)
 1.5. Forall y, (In(y, X) -> In(y, EmptySet)). (Universal Generalization from subproof below)
 begin subproof
  1.5.1. Let y be any object. (Arbitrary)
  1.5.2. In(y, X) -> In(y, EmptySet). (Conditional Introduction from subproof below)
  begin subproof
   1.5.2.1. In(y, X). (Assumption for Conditional Introduction)
   1.5.2.2. Not(In(y, X)). (from 1.2, 1.5.2.1)
   1.5.2.3. In(y, EmptySet). (Explosion from 1.5.2.1, 1.5.2.2)
  end subproof
 end subproof
 1.6. X is a subset of EmptySet. (Universal Instantiation from 1.5)
 1.7. Equals(X, EmptySet). (Set Equality from 1.4, 1.6)
end subproof
